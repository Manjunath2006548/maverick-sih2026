% ==================== 1. AIM ====================
\section{Aim of the Project}

\answerbox{What is PRISMA trying to achieve?}{
  \textbf{PRISMA} (Predictive Reliability-based Intelligent Screening and Monitoring Analysis) is a
  machine-learning system built for ISRO that detects \textbf{latent burn-in defects} in electronic
  components before they are screened into an operational payload.

  The project has three concrete aims:
  \begin{enumerate}[leftmargin=1.5em]
    \item \textbf{Catch what static limits miss.} A component that passes the absolute datasheet
          limits (e.g.\ Iddq $<$ 50\,$\mu$A) but drifts abnormally \emph{relative to its own lot}
          is flagged as a latent defect --- the defect that traditional static pass/fail screening
          lets escape.
    \item \textbf{Predict failure before it happens.} Use the burn-in \emph{time series}
          (values at 0h / 24h / 96h / 168h) to forecast the 168h value and reject parts whose
          predicted drift exceeds a statistically-derived \emph{safety slope}.
    \item \textbf{Be explainable and auditable.} Every verdict (REJECT / REVIEW / PASS) carries a
          6-step reasoning chain, per-method votes, lot statistics, and a downloadable
          acceptance/rejection \textbf{certificate} --- because a QA inspector must be able to
          justify every screening decision.
  \end{enumerate}
}

\noindent The problem statement describes the exact scenario we target: during Environmental Stress
Screening (ESS), components are burned in at 125\,$^{\circ}$C and screened by static pass/fail
limits, but physically strong / electrically weak parts drift out of tolerance and fail in the
field. PRISMA turns the burn-in \emph{trajectory} into an early-warning signal.

% ==================== 2. WHAT WE HAVE DONE ====================
\section{What We Have Done (Complete Implementation)}

\answerbox{The full scope of the delivered system}{
  We implemented PRISMA end-to-end --- from the ML core to the live deployment --- entirely
  \textbf{from scratch}, with \textbf{zero ML libraries} and \textbf{zero servers}. The whole system
  runs inside the browser. Concretely, we built:
}

\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Module A --- Dynamic Lot-Relative Outlier Detection.}
        For every \emph{lot} and every parametric column, we compute lot statistics (mean, std,
        median, quartiles) and flag components by \textbf{Z-Score} ($|z|>1.5$), \textbf{IQR fences}
        ($Q_{25}-1.5\,IQR$ / $Q_{75}+1.5\,IQR$), and \textbf{absolute datasheet limits}
        (iddq/leakage 50\,$\mu$A, delay 5\,ns, supply 120\,mA), merged by a 2-of-3 consensus into a
        per-parameter anomaly vote and a component risk score.
  \item \textbf{Module B --- Time-Series Drift Predictor.}
        Ridge regression (closed-form L2), Random Forest (100 CART trees) and Gradient Boosting
        (100 shallow trees, lr 0.1) --- \emph{all implemented from scratch, no scikit-learn} ---
        trained from \code{Value\_0h}+\code{Value\_24h} (optionally 96h) features to forecast
        \code{Value\_168h}, with 5-fold cross-validation, per-parameter MAE/RMSE/$R^2$, and a
        safety-slope early-rejection rule.
  \item \textbf{Comprehensive pipeline.} One call runs Module A + Module B and merges verdicts by
        priority \keys{REJECT} $>$ \keys{REVIEW} $>$ \keys{PASS} (union), verified with zero
        mismatched rows across all 200 components.
  \item \textbf{Explainability.} A 6-step reasoning chain per component: raw values $\rightarrow$
        lot statistics $\rightarrow$ per-method votes $\rightarrow$ drift forecast $\rightarrow$
        comparison with safety slope $\rightarrow$ final verdict, plus burn-in temperature and ground
        truth comparison where available.
  \item \textbf{Authentication and session management.} Register-first login (no seeded accounts),
        SHA-256 password hashing, email+timestamp+random bearer tokens, 401 auto-redirect, and a
        logout endpoint that invalidates the token.
  \item \textbf{Data ingestion.} CSV upload (hand-written parser: quoted fields, CRLF/LF, type
        coercion), manual \code{key:value} entry, and a deterministic sample-data generator
        (MT19937, seed 42) supporting \textbf{200 up to 100{,}000 components}.
  \item \textbf{Five-page ISRO-branded UI.} Landing, Login/Register, Upload, Analysis (4 tabs +
        explain panel + certificates), and a Mission-Control Dashboard.
  \item \textbf{Deployment and verification.} Static export deployed to
        \url{https://isro-burnin-detection.vercel.app}, verified by a 26-check E2E suite and a
        75-check deep audit; every deterministic number in this report is reproduced by those tests.
\end{itemize}

% ==================== 6. OUTPUT ANALYSIS ====================
\section{How the Output We Will Get (Analysis)}

\subsection*{6.1 Module A --- Outlier Detection output}
Each component gets: per-parameter \code{Z / IQR / absolute} votes (FLAG or CLEAR), a risk score,
and a classification:
\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Classification} & \textbf{Rule}\\
\midrule
\keys{REJECT} & risk $> 0.7$, or $> 50\%$ parameters anomalous\\
\keys{REVIEW} & risk $> 0.4$, or any parameter anomalous\\
\keys{PASS} & otherwise\\
\bottomrule
\end{tabular}
\end{center}
Plus a \textbf{detection score} that penalises false negatives 10$\times$:
\[
\text{Score}=\max\!\left(0,\;\frac{TP+TN-10\cdot FN-1\cdot FP}{N}\right)\times100\%
\]

\subsection*{6.2 Module B --- Drift Prediction output}
For each of the four parameters: trained model choice, MAE / RMSE / $R^2$ / MAPE, safety slope, and
feature importances; per component a \code{Value\_168h} forecast, a drift rate, the safety slope
comparison, and a \keys{PASS}/\keys{REJECT} recommendation with confidence.

\subsection*{6.3 Comprehensive output}
One merged verdict per component (\keys{REJECT} $>$ \keys{REVIEW} $>$ \keys{PASS}), Module A
summary, Module B summary, and per-component detail (all votes, predictions, reasoning chain).

\subsection*{6.4 Explainability output}
For any component (\code{GET /api/analysis/explain/COMP-01001}): raw values, lot statistics,
per-method votes, drift analysis per parameter, burn-in temperature, and the 6-step reasoning
chain.

\subsection*{6.5 Dashboard and certificates}
Quick stats (components / lots / anomalies / models), Module A verdict bars, Module B accuracy
cards, and downloadable HTML \emph{Acceptance / Rejection / Manual Review} certificates with the
8-item screening checklist and QA sign-off block.

% ==================== 11. BEHAVIOUR ====================
\section{How It Behaves (And Why It Behaves Like This)}

\subsection*{11.1 Behaviour}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Deterministic:} given the same input, every run returns identical outputs (seed 42).
  \item \textbf{Responds in real time:} 200 components analysed in well under a second; a
        10{,}000-component lot in $\sim$6.4\,s; heavy loops yield every 24 iterations to keep the UI
        responsive.
  \item \textbf{Guards edge cases:} zero std $\rightarrow$ $z=0$ (no false positives); empty dataset
        $\rightarrow$ graceful empty result; singular matrix $\rightarrow$ mean prediction;
        $<$10 rows $\rightarrow$ parameter skipped; localStorage full $\rightarrow$ caught;
        unknown parameter $\rightarrow$ HTTP 400.
  \item \textbf{Session behaviour:} wrong password / unregistered email $\rightarrow$ 401; logout
        invalidates the token; 401 auto-redirects and clears the stored user.
\end{itemize}

\subsection*{11.2 Why it behaves like this}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Lot-relative, not global, statistics} --- because a 45\,$\mu$A part in a 10\,$\mu$A
        lot is the actual defect the problem statement describes; global limits cannot see it.
  \item \textbf{FN penalised 10$\times$} --- because an escaped defect destroys a mission; false
        negatives are catastrophically more expensive than reviews.
  \item \textbf{Safety slope from statistics} ($|\bar{r}_{0\rightarrow24}|+1.96\,s_r$) rather than a
        hand-picked constant --- it adapts to each parameter and lot and is defensible as a
        97.5\%-two-sided confidence bound.
  \item \textbf{Consensus (2-of-3)} --- reduces false alarms while guaranteeing zero escapes.
  \item \textbf{In-browser execution} --- because data cannot leave the machine in defence use, and
        it makes the demo zero-deployment and instantly auditable.
  \item \textbf{From-scratch ML} --- reproducibility and full control over every numeric result;
        no black-box dependency could silently change the verified numbers.
\end{itemize}
