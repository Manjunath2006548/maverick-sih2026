% ==================== 1. ABSTRACT ====================
\section{Abstract and 100\% Conformance Statement}

\noindent In high-reliability sectors such as space, electronic components undergo rigorous
Environmental Stress Screening (ESS), including \textbf{Burn-In testing} --- operating components at
elevated temperatures (e.g.\ \textbf{125\,$^{\circ}$C}) for extended periods. Traditional screening
relies on \emph{static parametric pass/fail limits}. However, ``latent defects'' --- components that
pass the absolute datasheet limits yet exhibit subtle, anomalous drift over time --- often escape
into final payloads and fail catastrophically in the field.

\noindent \textbf{PRISMA} is a predictive machine-learning system that analyses time-series
parametric data (standby current \keys{Iddq}, leakage currents, propagation delays) measured at
the burn-in intervals \keys{0h, 24h, 96h, 168h} to detect anomalous components before they are
screened into an operational payload.

\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Module A --- Dynamic Outlier Detection:} Z-Score + IQR + absolute datasheet limits
        against the \emph{lot distribution} (not global) with 2-of-3 consensus. The
        10\,$\mu$A-lot / 45\,$\mu$A-part / 50\,$\mu$A-limit example is caught live.
  \item \textbf{Module B --- Time-Series Drift Predictor:} Ridge / Random Forest / Gradient
        Boosting (all from scratch, no ML libraries) from \code{Value\_0h}+\code{Value\_24h} to
        \code{Value\_168h}; safety-slope early rejection with 6-step reasoning chain.
  \item \textbf{Evaluation:} Anomaly Detection Score (FN penalised 10$\times$), MAE per parameter,
        Explainability for every verdict.
  \item \textbf{Combined verdict:} Module A $\cup$ Module B merged as
        \keys{REJECT}$>$\keys{REVIEW}$>$\keys{PASS}.
\end{itemize}

\whatbox{Every phrase of the problem statement maps to a shipped artefact: 125\,$^{\circ}$C
  burn-in as \code{burn\_in\_temp}; the 10/45/50\,$\mu$A anomaly caught by lot-relative
  consensus with pre-configured 50\,$\mu$A absolute limit; Module B consuming exactly
  \code{Value\_0h}+\code{Value\_24h} to forecast \code{Value\_168h}; FN penalised 10$\times$;
  MAE per parameter; and a full audit panel for QA inspectors.}

\section{The Six Phases (0\% $\rightarrow$ 100\%)}

% ==================== Phase 1 ====================
\phasebox{Phase 1}{0\% -- 15\%}{Problem Framing and Requirement Analysis}

\subsection*{Aim}
To build a predictive ML system that detects physically strong but electrically weak
components --- components that pass today's static datasheet pass/fail screening yet carry
\emph{latent defects} that will fail catastrophically during flight. PRISMA catches them during
burn-in so they never enter an operational payload.

\subsection*{Why we are doing this (motivation)}
\begin{itemize}[leftmargin=1.5em]
  \item In burn-in screening, a part is rejected only when a parameter crosses an absolute
        datasheet limit (e.g.\ 50\,$\mu$A Iddq). A part drifting 45\,$\mu$A in a 10\,$\mu$A lot
        still ships.
  \item Field failures of such ``escaped'' components are catastrophic: mission loss, payload
        failure, and loss of investment. One failed part in one spacecraft can destroy an entire
        launch.
  \item Burn-in generates \emph{time-series} data (0h/24h/96h/168h) that encodes the health
        trajectory of every component --- but static limits ignore the trajectory. The trend is
        the signal.
  \item Manual review of thousands of parts does not scale. A dynamic, lot-relative, explainable
        ML system scales and is auditable by QA inspectors.
\end{itemize}

\subsection*{Use case (what this system is for)}
A QA inspector at an ISRO test floor uploads burn-in parametric data for a lot of components.
PRISMA (i) flags lot-relative outliers with Z-score + IQR + absolute limits, (ii) forecasts the
168h parameter from the 0h/24h trend and rejects parts whose predicted drift exceeds a
statistically-derived safety slope, and (iii) returns a merged REJECT / REVIEW / PASS verdict
with a 6-step explanation chain and downloadable acceptance/rejection certificates. No defective
component escapes the floor.

\subsection*{Where we have implemented}
All logic exists in-browser (\code{lib/maverick-api.js}, 1780 lines) so the system runs with zero
servers and zero ML libraries --- fully from scratch --- and is deployed as a static site at
\url{https://isro-burnin-detection.vercel.app}. Source lives at
\url{https://github.com/Manjunath2006548/maverick-sih2026} (commit \code{aefd16c}).

\subsection*{Deliverable}
A complete, unambiguous reading of the problem statement into implementable requirements.

\subsection*{What was established}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Background:} 125\,$^{\circ}$C burn-in inside ESS; static pass/fail lets latent
        defects escape.
  \item \textbf{Description:} predict defective parts from time-series parametric data at
        0h/24h/96h/168h (\keys{iddq}, \keys{leakage}, \keys{delay}, \keys{supply\_current}).
  \item \textbf{Module A:} dynamic lot-relative outlier detection (10\,$\mu$A / 45\,$\mu$A /
        50\,$\mu$A example).
  \item \textbf{Module B:} regression from \code{Value\_0h}+\code{Value\_24h} to
        \code{Value\_168h} with safety-slope cutoff.
  \item \textbf{Metrics:} Anomaly Detection Score (FN-catastrophic), Drift Accuracy (MAE),
        Explainability.
\end{itemize}

\subsection*{Verified by}
26-check E2E suite + 75-check deep audit re-verify every number.

% ==================== Phase 3 ====================
\phasebox{Phase 3}{35\% -- 60\%}{Data Ingestion and Module A (Dynamic Outlier Detection)}

\subsection*{Aim}
Generate a deterministic, reproducible burn-in dataset and implement Module A: dynamic,
lot-relative outlier detection that catches the 10\,$\mu$A $\rightarrow$ 45\,$\mu$A $\rightarrow$
50\,$\mu$A defect that static datasheet screening misses.

\subsection*{Why we are doing this}
\begin{itemize}[leftmargin=1.5em]
  \item Every result in this report must be \textbf{reproducible by any evaluator}. Seed-42
        MT19937 guarantees the identical dataset for every run, everywhere.
  \item Real burn-in data is proprietary (ISRO test-floor records are classified). A
        physics-informed synthetic generator (base ranges, lot drift, defect multipliers, Gaussian
        noise, 125\,$^{\circ}$C burn-in temperature) produces realistic data for a live demo.
  \item Static 50\,$\mu$A limits cannot see lot-relative anomalies. A part at 45\,$\mu$A in a
        10\,$\mu$A lot is physically fine by datasheet but statistically impossible --- the
        trajectory is the tell-tale of a latent defect.
  \item Three independent detection methods (Z-score, IQR, absolute limit) with consensus
        reduce false alarms while guaranteeing zero escapes.
\end{itemize}

\subsection*{What this phase delivers}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{CSV upload:} hand-written parser (quoted fields, CRLF/LF, numeric/boolean/null
        coercion); Excel rejected with clear error.
  \item \textbf{Manual entry:} \code{key:value} lines queued per component.
  \item \textbf{Sample data:} default 200 components $\times$ 5 lots $\times$ 4 checkpoints =
        \textbf{20 columns}, $\sim$8\% defect rate, 15 seeded defects. The component count is
        configurable (default 200, supports large datasets up to 100{,}000 --- e.g.\ 10{,}000)
        via \code{/api/data/sample?count=N}.
  \item \textbf{Module A outlier analysis} with per-method verdicts, risk score, and consensus.
  \item \textbf{Detection score} with FN penalised 10$\times$.
\end{itemize}

\subsection*{Where we have implemented}
\code{lib/maverick-api.js}: \code{generateFlatData}, \code{analyzeOutliers},
\code{detectionScore} (Z-score/IQR/absolute methods), \code{requireAuth},

\subsection*{3.2 Module A --- three detection methods}
For every lot and every parametric column, lot statistics (mean, std, median, quartiles, min/max)
are computed. Each value is judged by:
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Z-Score:} $z=(x-\mu_{\text{lot}})/\sigma_{\text{lot}}$, anomaly if $|z|>1.5$.
        Guard: if $\sigma=0$, $z=0$ (no false positives on uniform lots).
  \item \textbf{IQR bounds:} $[Q_{25}-1.5\cdot IQR,\;Q_{75}+1.5\cdot IQR]$, anomaly if outside.
  \item \textbf{Absolute datasheet limits:} hard pass/fail. Defaults: \textbf{50\,$\mu$A}
        (\code{iddq}, \code{leakage}), \textbf{5\,ns} (\code{delay}), \textbf{120\,mA}
        (\code{supply\_current}). Forced on load so stale localStorage cannot override.
\end{enumerate}

\subsection*{3.3 Risk score and consensus}
Per-parameter risk: $\text{risk} = \bar{c}\times(1 + 0.2\times v)$, capped at 1.0, where $\bar{c}$
= mean confidence of 3 methods, $v$ = anomaly vote count. Component risk = mean of per-parameter
risks. Classification: risk$>0.7$ or $>50\%$ params anomalous $\rightarrow$ \keys{REJECT};
risk$>0.4$ or any anomaly $\rightarrow$ \keys{REVIEW}; else \keys{PASS}.

\subsection*{3.4 The 10\,$\mu$A $\rightarrow$ 45\,$\mu$A $\rightarrow$ 50\,$\mu$A case}
A part at 45\,$\mu$A in a lot whose mean is 10\,$\mu$A sits inside the 50\,$\mu$A datasheet wall.
Static screening ships it. PRISMA does not --- Z-score and IQR are computed against the lot mean,
so 45\,$\mu$A is several sigma away and flagged. The 50\,$\mu$A absolute limit also fires as a
third vote.

\subsection*{3.5 Detection score (FN penalised 10$\times$)}
\begin{equation*}
\text{Score}=\max\left(0,\;\frac{TP+TN-10\cdot FN-1\cdot FP}{N}\right)\times100\%
\end{equation*}

% ==================== Phase 4 ====================
\phasebox{Phase 4}{60\% -- 80\%}{Module B and Evaluation Metrics}

\subsection*{Aim}
Build Module B: a time-series drift predictor that forecasts the 168h parameter value from the
0h/24h (optionally 96h) trend, and rejects components whose predicted drift exceeds a
statistically-derived safety slope --- catching failures \emph{before they happen}.

\subsection*{Why we are doing this}
\begin{itemize}[leftmargin=1.5em]
  \item Module A sees the component \emph{today}. A component can be within every limit today yet
        be on a trajectory to fail by 168h. Module B forecasts that trajectory.
  \item The problem statement demands ``drift accuracy (MAE)'' --- Module B provides per-parameter
        MAE, RMSE, $R^2$, MAPE on a train/validate split.
  \item Implementing Ridge / Random Forest / Gradient Boosting \textbf{from scratch} (no scikit-learn)
        proves the ML fundamentals and guarantees zero external ML dependencies.
  \item A statistical safety slope (mean 0$\rightarrow$24h drift + 1.96$\cdot$std) is more
        defensible than a hand-picked constant threshold: it adapts to each parameter and lot.
\end{itemize}

  \item \textbf{Feature engineering} (per parameter, per component): value at 0h/24h/(96h), drift,
        drift rate, percentage change, acceleration; \code{has96} flag adapts vector length.
  \item \textbf{Three regressors:} Ridge (closed-form L2), RF (100 CART trees), GB (100 shallow
        trees) --- with StandardScaler and 5-fold sequential CV for model selection.
  \item \textbf{Safety-slope early rejection} with margin and confidence.
  \item \textbf{Drift Prediction Accuracy} table (MAE/RMSE/$R^2$) for all four parameters.
\end{itemize}

\subsection*{4.1 Feature engineering}
Per parameter, per component: \code{value\_0h}, \code{value\_24h}, drift
($x_{24h}-x_{0h}$), drift rate ($(x_{24h}-x_{0h})/24$), percentage change
($100\cdot\Delta/|x_{0h}|$). When 96h exists: additionally \code{value\_96h}, drift$_{24h\rightarrow96h}$,
drift rate$_{24h\rightarrow96h}$, acceleration. The \code{has96} flag adapts the feature vector
length. Target: \code{Value\_168h}. Minimum 10 training samples required (parameters with fewer
are skipped).

\subsection*{4.2 Three regressors (from scratch, no ML libraries)}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Ridge regression}: closed-form L2 ($\alpha=1.0$), Gauss-Jordan elimination with
        partial pivoting (threshold $10^{-12}$); falls back to mean prediction if singular.
  \item \textbf{Random Forest}: 100 CART trees, max depth 10, bootstrap (resample $n$ indices with
        replacement), \textbf{all features} at each split (not sqrt($d$)), leaf value = mean($y$),
        MSE reduction splitting criterion: $\text{loss} = \text{leftSq} - \text{leftSum}^2/\text{leftN}
        + \text{rightSq} - \text{rightSum}^2/\text{rightN}$.
  \item \textbf{Gradient Boosting}: 100 shallow trees (depth 5, lr 0.1), initial prediction
        $\bar{y}$, sequential residual fitting. Feature importance: MSE improvement at each split.
\end{itemize}

\subsection*{4.3 StandardScaler}
Z-score standardisation: $z = (x - \mu)/\sigma$ using \textbf{population std} (not sample std).
Guard: if $\sigma=0$, $z=0$.

\subsection*{4.4 Cross-validation}
5-fold sequential CV (not shuffled, not stratified), dynamic fold count
$\min(5, \lfloor n/2 \rfloor)$. Scoring: negative MAE. Best model selected by lowest mean
CV MAE.

\subsection*{4.5 Prediction and early rejection}
Given \code{Value\_0h} and \code{Value\_24h} (optionally \code{Value\_96h}):
\begin{enumerate}[leftmargin=1.5em]
  \item Build and standardise feature vector (length adapts via \code{has96}).
  \item Predict $\hat{x}_{168h}$ with best model.
  \item Drift rate $r=(\hat{x}_{168h}-x_{0h})/168$.
  \item Safety slope $s$: $|r|>s$ $\rightarrow$ \keys{REJECT}; else \keys{PASS} with confidence
        $1-|r|/s$.
\end{enumerate}
Unknown parameter returns HTTP 400 (not a silent 200-with-error).

\subsection*{4.6 Safety slope}
\begin{equation*}
s =
\begin{cases}
|\overline{r}_{0\rightarrow24}|+1.96\cdot s_r, & n\ge 5\\
2\times|\overline{r}_{0\rightarrow24}|, & n<5
\end{cases}
\end{equation*}
where $\Phi^{-1}(0.975)=1.959963984540054$. Margin: $s - |r|$.

  \item \keys{Upload Data} (sidebar $\rightarrow$ \code{/upload}).
  \item Pick one of three tabs:
    \begin{itemize}[leftmargin=1.5em]
      \item \keys{Sample Data} --- one click generates the deterministic seed-42 dataset
            (default 200 components, 5 lots, 20 columns, $\sim$8\% defect rate); the component
            count field supports large datasets up to 100{,}000 (e.g.\ 10{,}000). Recommended for a demo.
      \item \keys{CSV Upload} --- drag a \code{.csv}; Excel files are rejected with a clear
            message. Preview shows the first 10 rows.
      \item \keys{Manual Data} --- type \code{key: value} lines per component and queue them.
    \end{itemize}
  \item Once ready a green banner appears: ``Data loaded (200 components / 5 lots)''.
\end{enumerate}

\subsection*{Step 4 --- Run the analysis}
Navigate to \keys{Analysis} (\code{/analysis}) and select a tab:
\begin{enumerate}[leftmargin=1.5em]
  \item \keys{Comprehensive} (recommended) --- runs Module A + Module B together,
        trains all four parameter models, and shows the merged verdict summary.
  \item \keys{Module A} --- lot-relative outlier detection only.
  \item \keys{Module B} --- drift training + batch prediction with per-parameter
        MAE / RMSE / $R^2$ cards.
  \item \keys{Single Prediction} --- pick a parameter and a component (or enter
        \code{Value\_0h}/\code{Value\_24h}/\code{Value\_96h}) to get the 168h forecast and a
        PASS/REJECT recommendation. Example verified number: iddq $\rightarrow$
        \code{Value\_168h = 11.4747}, PASS.
\end{enumerate}

\subsection*{Step 5 --- Read the results}
\begin{enumerate}[leftmargin=1.5em]
  \item The summary shows \keys{PASS} (green, e.g.\ 164), \keys{REVIEW} (orange, 33) and
        \keys{REJECT} (red, 3). Each count is a \textbf{clickable filter}.
  \item Click any component row to open the classification detail overlay:
    Module A votes per parameter (Z / IQR / absolute = FLAG or CLEAR), Module B prediction vs
    safety slope (Rm $\cdot$ 1.96), and the \textbf{6-step reasoning chain}.
  \item Click \keys{Explain} in the navigation for the full explainability panel of any
        component (e.g.\ \code{COMP-01001}).
  \item Download the \keys{Acceptance / Rejection / Manual Review} HTML certificate. The
        certificate includes the 8-item screening checklist and the QA sign-off block.
\end{enumerate}

\subsection*{Step 6 --- Monitor with the dashboard}
\keys{Dashboard} shows quick stats (Components / Lots / Anomalies / Models), Module A verdict
bars, Module B accuracy cards, and the 4-step system pipeline visual. Accessible to every role.

\subsection*{Step 7 --- Sign out and secure the session}
\keys{Sign Out} in the sidebar calls \code{POST /api/auth/logout} and destroys the token.
